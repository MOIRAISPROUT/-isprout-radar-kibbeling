genereer signalen waar keywords vallen.""" feed = feedparser.parse( f"https://news.google.com/rss/search?q={query.replace(' ', '+')}&hl=nl&gl=NL&ceid=NL:nl" ) out = [] for e in feed.entries[:10]: text = e.title + " " + getattr(e, "summary", "") hits = score_text(text, keywords) if hits: out.append({ "type": "A" if keywords is OUT_KEYWORDS else "B", "titel": e.title[:140], "link": e.link, "bron": bron, "gewicht": sum(h["gewicht"] for h in hits), "datum": getattr(e, "published", ""), }) return out leads = [] for k in KANTOREN: sig_a = scan(f'{k["name"]} accountants nieuws', OUT_KEYWORDS, "Google News") sig_b = scan(f'"{k["name"]}" vacature', NL_KEYWORDS, "Google News (vacatures)") # basis + signaal impact → score 0-100 score_a = min(100, 30 + sum(s["gewicht"] for s in sig_a) * 6) score_b = min(100, 30 + sum(s["gewicht"] for s in sig_b) * 6) leads.append({ "name": k["name"], "city": k["city"], "score_out": score_a, "score_nl": score_b, "signalen": sig_a + sig_b, }) out = { "gegenereerd": datetime.datetime.now().isoformat(), "bron_verklaring": "Score = startwaarde + gewogen keyword-trefters. Demo-kwaliteit: geen echte AI-interpretering.", "leads": leads, } with open("leads.json", "w", encoding="utf-8") as f: json.dump(out, f, ensure_ascii=False, indent=2) print(f"{len(leads)} kantoren gescand — scores A (outsourcing) en B (NL-plaatsing)")
